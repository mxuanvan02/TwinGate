\newcommand{\fittocol}[1]{%
  \begin{lrbox}{\fitbox}#1\end{lrbox}%
  \ifdim\wd\fitbox>\columnwidth
    \resizebox{\columnwidth}{!}{\usebox{\fitbox}}%
  \else
    \usebox{\fitbox}%
  \fi}
